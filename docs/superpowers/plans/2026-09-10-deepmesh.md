# DeepMesh 可微等值面提取 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在当前 DeepSDF 仓库中实现 DeepMesh 论文（arXiv:2106.11795v2）的可微等值面提取：分层 octree 采样 + marching cubes 前向 + 基于隐函数定理（Theorem 1）的顶点梯度回传，并提供网格级 latent 优化脚本。

**Architecture:** 前向用 octree 采样 + skimage marching cubes 提取网格（每次前向重新提取，拓扑可变）；用 autograd 求顶点解析法向 `n=∇f(v)`；通过恒等式 `v_diff = v − (n/‖n‖²)·(f(z,v) − f(z,v).detach())` 把顶点重新挂回 autograd 图，其梯度恰为 Theorem 1 的 `∂x/∂c = −(n/‖n‖²)·∂f/∂c`，普通 autograd 即可同时支持 `c=z` 与 `c=[z|Θ]`。

**Tech Stack:** PyTorch（autograd）、scikit-image（marching cubes，复用现有依赖）、plyfile、numpy。

**设计文档:** `docs/superpowers/specs/2026-09-10-deepmesh-design.md`

- [ ] **Step 1: 编写 `deep_sdf/differentiable_mesh.py`**

要点：
- octree 细分准则（论文 Algorithm 1）：体素 8 角点 `min|f| < threshold_factor × voxel_size` 则细分；分辨率链 `N → 2N−1`（粗格点是细格子集）；未评估的细格点用粗网格三线性插值填充（未细分体素 8 角点同号且 `|f|≥阈值`，凸组合保号、保下界，不会产生伪零交叉——因为 SDF 1-Lipschitz，体素对角线 √3Δx < 2Δx）。
- IFT 重连恒等式实现 Theorem 1。
- 空网格（无零交叉）：捕获异常/空结果，warning 后返回空张量。
- 法向求解显式 `torch.enable_grad()`，兼容外部 `no_grad` 上下文。

```python
#!/usr/bin/env python3
# Copyright 2004-present Facebook. All Rights Reserved.

import logging

import numpy as np
import plyfile
import torch
import torch.nn.functional as F

try:
    from skimage.measure import marching_cubes
except ImportError:
    from skimage.measure import marching_cubes_lewiner as marching_cubes

import deep_sdf.utils


def _eval_points(decoder, latent_vec, coords, max_batch):
    """ coords: (M,3) torch tensor on CPU; returns (M,) SDF values on CPU. """
    num_samples = coords.shape[0]
    values = torch.zeros(num_samples)
    head = 0
    while head < num_samples:
        subset = coords[head : min(head + max_batch, num_samples)].cuda()
        values[head : min(head + max_batch, num_samples)] = (
            deep_sdf.utils.decode_sdf(decoder, latent_vec, subset)
            .squeeze(1)
            .detach()
            .cpu()
        )
        head += max_batch
    return values


def _grid_coords(N):
    """ Row-major (i,j,k) grid point coordinates over [-1, 1]^3, (N^3, 3). """
    voxel_size = 2.0 / (N - 1)
    overall_index = torch.arange(0, N ** 3, out=torch.LongTensor())
    samples = torch.zeros(N ** 3, 3)
    samples[:, 2] = overall_index % N
    samples[:, 1] = torch.div(overall_index, N, rounding_mode="floor") % N
    samples[:, 0] = torch.div(overall_index, N ** 2, rounding_mode="floor") % N
    return samples * voxel_size - 1.0


def sample_sdf_octree(
    decoder,
    latent_vec,
    resolution,
    max_batch,
    initial_resolution=32,
    threshold_factor=2.0,
):
    """ Hierarchically sample the SDF on an octree-refined grid (DeepMesh Alg. 1).

    Voxels are subdivided when the smallest absolute corner value is below
    threshold_factor * voxel_size (the voxel diagonal criterion). Unevaluated
    fine-grid points are filled by trilinear interpolation of the coarse grid,
    which is sign-consistent and cannot introduce spurious zero crossings.

    Returns (sdf, N): sdf is a (N,N,N) FloatTensor, N the reached resolution
    (N follows the chain initial -> 2N-1, so it may exceed `resolution`).
    """
    decoder.eval()

    N = initial_resolution
    sdf = _eval_points(decoder, latent_vec, _grid_coords(N), max_batch).reshape(
        N, N, N
    )

    while N < resolution:
        voxel_size = 2.0 / (N - 1)
        threshold = threshold_factor * voxel_size

        a = sdf.abs()
        cell_min = torch.minimum(
            torch.minimum(
                torch.minimum(a[:-1, :-1, :-1], a[1:, :-1, :-1]),
                torch.minimum(a[:-1, 1:, :-1], a[1:, 1:, :-1]),
            ),
            torch.minimum(
                torch.minimum(a[:-1, :-1, 1:], a[1:, :-1, 1:]),
                torch.minimum(a[:-1, 1:, 1:], a[1:, 1:, 1:]),
            ),
        )
        split = cell_min < threshold  # (N-1)^3

        if not split.any():
            logging.debug("octree: no voxel to split at resolution {}".format(N))
            break

        N_fine = 2 * N - 1

        # trilinear fill reproduces coarse values at even indices exactly
        fine = (
            F.interpolate(
                sdf.reshape(1, 1, N, N, N),
                size=(N_fine, N_fine, N_fine),
                mode="trilinear",
                align_corners=True,
            )
            .reshape(N_fine, N_fine, N_fine)
        )

        # a fine point needs evaluation if any adjacent coarse cell is split
        dilated = (
            F.max_pool3d(
                split.float().reshape(1, 1, N - 1, N - 1, N - 1),
                kernel_size=3,
                stride=1,
                padding=1,
            ).reshape(N - 1, N - 1, N - 1)
            > 0
        )
        # padded cell array so that cell index ci -> ci + 1 is always valid
        pad = torch.zeros(N + 1, N + 1, N + 1, dtype=torch.bool)
        pad[1:N, 1:N, 1:N] = dilated

        fine_index = torch.arange(N_fine)
        required = torch.zeros(N_fine, N_fine, N_fine, dtype=torch.bool)
        for di in (0, 1):
            for dj in (0, 1):
                for dk in (0, 1):
                    ii = fine_index // 2 - di + 1
                    jj = fine_index // 2 - dj + 1
                    kk = fine_index // 2 - dk + 1
                    required |= pad[ii][:, jj][:, :, kk]

        odd = (fine_index % 2) == 1
        new_points = required & (
            odd.reshape(-1, 1, 1) | odd.reshape(1, -1, 1) | odd.reshape(1, 1, -1)
        )

        indices = new_points.nonzero()  # (M, 3)
        if indices.shape[0] > 0:
            voxel_size_fine = 2.0 / (N_fine - 1)
            coords = indices.float() * voxel_size_fine - 1.0
            values = _eval_points(decoder, latent_vec, coords, max_batch)
            fine[new_points] = values

        sdf = fine
        N = N_fine
        logging.debug(
            "octree: refined to {}, evaluated {} new points".format(
                N, indices.shape[0]
            )
        )

    return sdf, N


def compute_sdf_gradients(decoder, latent_vec, points, max_batch=2 ** 18):
    """ Analytic gradient of the network output w.r.t. xyz at each point. """
    decoder.eval()
    grads = []
    head = 0
    num_points = points.shape[0]
    while head < num_points:
        chunk = (
            points[head : min(head + max_batch, num_points)]
            .detach()
            .clone()
            .requires_grad_(True)
        )
        with torch.enable_grad():
            sdf = deep_sdf.utils.decode_sdf(decoder, latent_vec, chunk)
            grad = torch.autograd.grad(sdf.sum(), chunk)[0]
        grads.append(grad.detach())
        head += max_batch
    return torch.cat(grads, 0)


def extract_differentiable_mesh(
    decoder,
    latent_vec,
    resolution=128,
    initial_resolution=32,
    max_batch=2 ** 18,
    threshold_factor=2.0,
    eps=1e-8,
):
    """ Extract the zero level set as a mesh whose vertices are differentiable
    with respect to the latent code (and decoder weights) via the implicit
    function theorem (DeepMesh Theorem 1):

        dx/dc = -(n / ||n||^2) * df/dc,   n = grad_x f(z, x)

    Returns (verts, faces, normals): verts (M,3) carries gradients, faces
    (F,3) is an integer index tensor, normals (M,3) are detached.
    """
    sdf, N = sample_sdf_octree(
        decoder,
        latent_vec,
        resolution,
        max_batch,
        initial_resolution,
        threshold_factor,
    )
    voxel_size = 2.0 / (N - 1)

    try:
        verts_np, faces_np, _, _ = marching_cubes(
            sdf.numpy(), level=0.0, spacing=(voxel_size,) * 3
        )
    except (ValueError, RuntimeError) as e:
        logging.warning("iso-surface extraction failed: {}".format(e))
        empty = torch.zeros(0, 3)
        return empty.cuda(), torch.zeros(0, 3).long().cuda(), empty.cuda()

    verts = torch.from_numpy(verts_np.copy()).float().cuda() - 1.0
    faces = torch.from_numpy(faces_np.copy()).long().cuda()

    normals = compute_sdf_gradients(decoder, latent_vec, verts, max_batch)

    # reattachment identity: value equals verts, gradient is Theorem 1
    sdf_at_verts = deep_sdf.utils.decode_sdf(decoder, latent_vec, verts)
    normals_sq = (normals ** 2).sum(-1, keepdim=True).clamp_min(eps)
    verts = verts - normals / normals_sq * (sdf_at_verts - sdf_at_verts.detach())

    return verts, faces, normals


def save_mesh(verts, faces, filename):
    """ Write a (verts, faces) mesh to a ply file. """
    verts = verts.detach().cpu().numpy()
    faces = faces.detach().cpu().numpy()

    num_verts = verts.shape[0]
    num_faces = faces.shape[0]

    verts_tuple = np.zeros((num_verts,), dtype=[("x", "f4"), ("y", "f4"), ("z", "f4")])
    for i in range(0, num_verts):
        verts_tuple[i] = tuple(verts[i, :])

    faces_building = []
    for i in range(0, num_faces):
        faces_building.append(((faces[i, :].tolist(),)))
    faces_tuple = np.array(faces_building, dtype=[("vertex_indices", "i4", (3,))])

    el_verts = plyfile.PlyElement.describe(verts_tuple, "vertex")
    el_faces = plyfile.PlyElement.describe(faces_tuple, "face")

    ply_data = plyfile.PlyData([el_verts, el_faces])
    logging.debug("saving mesh to %s" % filename)
    ply_data.write(filename)

def chamfer_distance(p, q, chunk=2048):
    """ Two-sided Chamfer distance, computed in chunks over q. """

    def min_dist(a, b):
        dmin = torch.full((a.shape[0],), float("inf"), device=a.device)
        for i in range(0, b.shape[0], chunk):
            d = torch.cdist(a, b[i : i + chunk])
            dmin = torch.minimum(dmin, d.min(dim=1).values)
        return dmin

    return min_dist(p, q).mean() + min_dist(q, p).mean()


def sample_mesh_surface(verts, faces, num_points):
    """ Uniformly sample points on a mesh surface, weighted by triangle area. """
    v = verts[faces]  # (F, 3, 3)
    areas = 0.5 * np.linalg.norm(
        np.cross(v[:, 1] - v[:, 0], v[:, 2] - v[:, 0]), axis=1
    )
    probs = areas / areas.sum()
    fi = np.random.choice(faces.shape[0], size=num_points, p=probs)
    u = np.sqrt(np.random.rand(num_points, 1))
    w = np.random.rand(num_points, 1)
    points = (1 - u) * v[fi, 0] + u * (1 - w) * v[fi, 1] + u * w * v[fi, 2]
    return points.astype(np.float32)


def adjust_learning_rate(initial_lr, optimizer, iteration, decreased_by=10, adjust_lr_every=None):
    lr = initial_lr * ((1 / decreased_by) ** (iteration // adjust_lr_every))
    for param_group in optimizer.param_groups:
        param_group["lr"] = lr


def optimize_latent_mesh(
    decoder,
    latent_init,
    target_points,
    num_iterations,
    resolution,
    lr,
    num_surface_samples=10000,
    log_every=50,
):
    """ Optimize a latent code against a surface point cloud with a Chamfer
    loss defined on the differentiably extracted mesh (DeepMesh Alg. 2).
    The mesh is re-extracted every iteration, so topology may change.
    """
    latent = latent_init.detach().clone().cuda().requires_grad_(True)
    target = target_points.detach().cuda()

    optimizer = torch.optim.Adam([latent], lr=lr)
    adjust_lr_every = int(num_iterations / 2)

    loss_num = 0.0
    empty_streak = 0
    for e in range(num_iterations):
        adjust_learning_rate(lr, optimizer, e, adjust_lr_every=adjust_lr_every)

        optimizer.zero_grad()

        verts, faces, _ = deep_sdf.differentiable_mesh.extract_differentiable_mesh(
            decoder, latent, resolution=resolution
        )

        if verts.shape[0] == 0:
            empty_streak += 1
            logging.warning("empty mesh at iteration {}".format(e))
            if empty_streak > 10:
                raise RuntimeError(
                    "iso-surface vanished; retry with --init sdf or a smaller --lr"
                )
            continue
        empty_streak = 0

        if verts.shape[0] > num_surface_samples:
            sel = torch.randperm(verts.shape[0])[:num_surface_samples]
            verts_sampled = verts[sel]
        else:
            verts_sampled = verts

        loss = chamfer_distance(verts_sampled, target)
        loss.backward()
        optimizer.step()

        if e % log_every == 0:
            logging.info("iter {} chamfer loss: {}".format(e, loss.item()))
        loss_num = loss.item()

    return loss_num, latent

            continue

        mesh_filename = os.path.join(reconstruction_meshes_dir, npz[:-4])
        latent_filename = os.path.join(
            reconstruction_codes_dir, npz[:-4] + ".pth"
        )
        if (
            args.skip
            and os.path.isfile(mesh_filename + ".ply")
            and os.path.isfile(latent_filename)
        ):
            continue

        logging.info("reconstructing {}".format(npz))

        full_filename = os.path.join(args.data_source, ws.sdf_samples_subdir, npz)
        data_sdf = deep_sdf.data.read_sdf_samples_into_ram(full_filename)
        data_sdf[0] = data_sdf[0][torch.randperm(data_sdf[0].shape[0])]
        data_sdf[1] = data_sdf[1][torch.randperm(data_sdf[1].shape[0])]

        # target point cloud: implicit-domain latent fit -> dense mesh ->
        # uniform surface samples
        _, target_latent = reconstruct(
            decoder, 800, latent_size, data_sdf, 0.01, 0.1,
            num_samples=8000, lr=5e-3, l2reg=True,
        )
        with torch.no_grad():
            target_verts, target_faces, _ = (
                deep_sdf.differentiable_mesh.extract_differentiable_mesh(
                    decoder, target_latent, resolution=256
                )
            )
        target_points = torch.from_numpy(
            sample_mesh_surface(
                target_verts.cpu().numpy(), target_faces.cpu().numpy(), 10000
            )
        )

        if args.init == "sdf":
            _, latent_init = reconstruct(
                decoder, 200, latent_size, data_sdf, 0.01, 0.1,
                num_samples=8000, lr=5e-3, l2reg=True,
            )
        else:
            latent_init = torch.ones(1, latent_size).normal_(mean=0, std=0.01)

        start = time.time()
        err, latent = optimize_latent_mesh(
            decoder,
            latent_init,
            target_points,
            int(args.iterations),
            args.resolution,
            args.lr,
        )
        logging.info("mesh optimization time: {}".format(time.time() - start))
        logging.info("final chamfer loss: {}".format(err))

        with torch.no_grad():
            verts, faces, _ = (
                deep_sdf.differentiable_mesh.extract_differentiable_mesh(
                    decoder, latent, resolution=256
                )
            )
        deep_sdf.differentiable_mesh.save_mesh(verts, faces, mesh_filename + ".ply")
        torch.save(latent.unsqueeze(0).detach().cpu(), latent_filename)
```

注：`with torch.no_grad()` 包住仅用于提取/保存的调用时，`extract_differentiable_mesh` 内的法向计算已显式 `enable_grad`，重连表达式在 no_grad 下退化为恒等（数值不变），行为正确。

- [ ] **Step 1: 编写 `/tmp/verify_topology.py`**

小 auto-decoder（latent=8，4 层 ×64，复用 `networks.deep_sdf_decoder.Decoder`）过拟合球（genus 0，χ=2）与环面（genus 1，χ=0）两个解析 SDF；然后从球的 latent 出发，用网格级 Chamfer 优化到环面点云，跟踪 Euler 示性数 χ = V − E + F：

```python
import sys
sys.path.insert(0, "/home/siqi/CLionProjects/DeepSDF")
import numpy as np
import torch

import deep_sdf
from networks.deep_sdf_decoder import Decoder
from reconstruct_mesh import (
    chamfer_distance,
    optimize_latent_mesh,
    sample_mesh_surface,
)

torch.manual_seed(0)
np.random.seed(0)


def sphere_sdf(x):
    return torch.norm(x, dim=1) - 0.5


def torus_sdf(x, R=0.4, r=0.15):
    q = torch.stack([torch.norm(x[:, 0:2], dim=1) - R, x[:, 2]], dim=1)
    return torch.norm(q, dim=1) - r


def euler_characteristic(faces):
    F = faces.shape[0]
    edges = set()
    for tri in faces.cpu().numpy():
        for a, b in ((0, 1), (1, 2), (2, 0)):
            e = (min(tri[a], tri[b]), max(tri[a], tri[b]))
            edges.add(e)
    return None  # placeholder, computed in main from verts count


latent_size = 8
decoder = Decoder(
    latent_size=latent_size,
    dims=[64, 64, 64, 64],
    latent_in=[2],
).cuda()

latents = torch.zeros(2, latent_size).normal_(0, 0.01).cuda().requires_grad_(True)

# overfit on analytic SDF samples of sphere and torus
opt = torch.optim.Adam(
    [{"params": decoder.parameters()}, {"params": latents, "lr": 1e-3}], lr=1e-4
)
for it in range(3000):
    pts = torch.rand(2, 2048, 3).cuda() * 2 - 1
    gt = torch.stack([sphere_sdf(pts[0]), torus_sdf(pts[1])]).unsqueeze(-1)
    gt = torch.clamp(gt, -0.1, 0.1)
    inp = torch.cat([latents.unsqueeze(1).expand(-1, 2048, -1), pts], dim=-1)
    pred = torch.clamp(decoder(inp.reshape(-1, latent_size + 3)), -0.1, 0.1)
    loss = torch.abs(pred - gt.reshape(-1, 1)).mean() + 1e-4 * latents.pow(2).mean()
    opt.zero_grad()
    loss.backward()
    opt.step()
print("overfit loss:", loss.item())

# target: torus surface samples
with torch.no_grad():
    tv, tf, _ = deep_sdf.differentiable_mesh.extract_differentiable_mesh(
        decoder, latents[1:2].detach(), resolution=63
    )
target_points = torch.from_numpy(
    sample_mesh_surface(tv.cpu().numpy(), tf.cpu().numpy(), 5000)
)

# mesh-level optimization from the sphere latent towards the torus
latent = latents[0:1].detach().clone().cuda().requires_grad_(True)
optimizer = torch.optim.Adam([latent], lr=1e-3)
for e in range(120):
    verts, faces, _ = deep_sdf.differentiable_mesh.extract_differentiable_mesh(
        decoder, latent, resolution=63
    )
    if verts.shape[0] == 0:
        print("iter {}: empty mesh".format(e))
        continue
    if verts.shape[0] > 5000:
        sel = torch.randperm(verts.shape[0])[:5000]
        verts = verts[sel]
    loss = chamfer_distance(verts, target_points.cuda())
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
    if e % 20 == 0 or e == 119:
        edges = set()
        f_np = faces.cpu().numpy()
        for tri in f_np:
            for a, b in ((0, 1), (1, 2), (2, 0)):
                edges.add((min(tri[a], tri[b]), max(tri[a], tri[b])))
        chi = verts.shape[0] - len(edges) + faces.shape[0]
        print(
            "iter {} chamfer {:.6f} chi {} (sphere=2, torus=0)".format(
                e, loss.item(), chi
            )
        )
```
